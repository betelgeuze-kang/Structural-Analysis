"""Source correspondence only; never fit or run a structural solver."""
from pathlib import Path
from decimal import Decimal
import hashlib
import json
import shutil
import time
import xml.etree.ElementTree as ET

root = Path(Path('/tmp/structural-soesianawati-primary-root.txt').read_text())
prior = Path('/mnt/193005ba-8531-4d0b-87c2-43c01ee2ce25/structural-soesianawati-intake-4wobdfzh')
started = time.perf_counter()
def digest(raw):
    return hashlib.sha256(raw).hexdigest()

raw = prior.with_name(prior.name + '.inventory.json').read_bytes()
assert digest(raw) == 'c69e062ac70ae8a674b52ab7ff72a16508772689d0e0d4e0f66352a565481e30'
inv = json.loads(raw)
entries = {r['path']: r for r in inv['files']}
bindings = []
for name in ['peer-7.xml', 'peer-8.xml', 'peer-9.xml', 'peer-10.xml', 'peer-7-history.txt']:
    data = (prior / name).read_bytes()
    assert digest(data) == entries[name]['sha256'] and len(data) == entries[name]['bytes']
    (root / name).write_bytes(data)
    bindings.append(entries[name])
assert digest((root / 'nist-5984.pdf').read_bytes()) == '2dbd0b88a3e8578c30f3de5e4b2728eb157b389d2d495be8f153109d8a951f0c'

# Human transcription of the visually reviewed table on printed p.7 / PDF p.15.
# This script verifies correspondence, not the correctness of that transcription.
table = [
    (7, '46.5', '744', '0.1', 5, '7', '85', 6, '170', '446', '364', '0.0086'),
    (8, '44.0', '2112', '0.3', 8, '8', '78', 5, '156', '446', '360', '0.0122'),
    (9, '44.0', '2112', '0.3', 7, '7', '91', 5, '182', '446', '364', '0.0080'),
    (10, '40.0', '1920', '0.3', 7, '6', '94', 5, '186', '446', '255', '0.0057'),
]
cohort = []
checks = 0
for pid, fc, axial, ratio, sets, dia, spacing, outer_sets, outer_spacing, fy, fyt, rho in table:
    e = ET.fromstring((root / f'peer-{pid}.xml').read_bytes()).find('columnTest')
    expected = {
        'materialProperties/concreteStrength': fc,
        'loading/axialLoad': axial,
        'materialProperties/longitudinalSteel/corner/yieldStress': fy,
        'materialProperties/longitudinalSteel/intermediate/yieldStress': fy,
        'materialProperties/transverseSteel/yieldStress': fyt,
        'geometry/width': '400', 'geometry/depth': '400',
        'geometry/lInflection': '1600', 'geometry/lMeasured': '1600',
        'longitudinalReinforcement/numberOfBars': '12',
        'longitudinalReinforcement/diameter': '16',
        'longitudinalReinforcement/reinforcementRatio': '0.0151',
        'transverseReinforcement/closeSpacing/numberHoopSets': str(sets),
        'transverseReinforcement/closeSpacing/barDiameter': dia,
        'transverseReinforcement/closeSpacing/hoopSpacing': spacing,
    }
    for path, value in expected.items():
        assert Decimal(e.findtext(path).replace(',', '')) == Decimal(value), (pid, path)
        checks += 1
    assert e.findtext('geometry/configuration') == 'Double Ended'
    assert e.findtext('loading/PDCode') == 'Shear provided'
    assert Decimal(axial) * 1000 / (Decimal(fc) * 400 * 400) == Decimal(ratio)
    cohort.append(dict(peer_id=pid, specimen=pid-6, matched_fields=expected,
                       nist_outer_hoop_sets=outer_sets, nist_outer_spacing_mm=outer_spacing,
                       nist_transverse_ratio=rho,
                       peer_transverse_ratio=e.findtext('transverseReinforcement/volTransReinfRatio'),
                       ratios_automatically_reconciled=False))

lines = (root / 'peer-7-history.txt').read_text().splitlines()
count = int(lines[1].strip()); pairs = [tuple(map(Decimal, line.split())) for line in lines[2:] if line.strip()]
assert count == len(pairs) == 744 and all(len(row) == 2 for row in pairs)
assert all(v.is_finite() for pair in pairs for v in pair)
unit1 = ET.fromstring((root / 'peer-7.xml').read_bytes()).find('columnTest')
damage = {c.tag: {'value': c.text, 'units': c.attrib.get('units')} for c in unit1.find('damage')}

summary = dict(
    schema='soesianawati-nist-source-correspondence.v1',
    nist_pdf_sha256=digest((root / 'nist-5984.pdf').read_bytes()),
    reviewed_pdf_pages=[9,11,15,49,53,105], reviewed_printed_pages=[1,3,7,41,45,97],
    source_role='NIST dataset compiler; secondary account of the 1986 physical experiment',
    original_1986_fulltext_retrieved=False,
    numeric_metadata_comparisons=checks, cohort=cohort,
    equivalent_cantilever=dict(length_m='1.6',
        force='NIST tabulated H is half central actuator force 2H; do not halve H again',
        displacement='central stub displacement relative to end supports; do not halve',
        axial_load_kn='744', axial_load_is_measured_history=False,
        p_delta_correction_inferred=False,
        exact_nist_disk_to_peer_file_identity_verified=False),
    unit1_history=dict(pairs=count, displacement_mm=[str(min(x for x,y in pairs)),str(max(x for x,y in pairs))],
                      force_kn=[str(min(y for x,y in pairs)),str(max(y for x,y in pairs))],
                      raw_sensor_status='not authenticated', actuator_command_status='not authenticated'),
    cover=dict(nist_clear_cover_to_transverse_bar_mm='13',
               peer_xml_location='longitudinalReinforcement/*/clearCover',
               longitudinal_bar_centres_inferred=False),
    unit1_damage_metadata=damage,
    source_limitations=[
        'NIST p.1 says most histories digitized from published plots; No.1 acquisition route is not individually identified.',
        'NIST p.3 warns transverse ratio definitions differ; preserve published ratios, derive model confinement only from authenticated geometry.',
        'NIST p.41 says 93 specimens whereas p.1 and p.97 say 107; no corpus count is admitted from these totals.',
        'Original disk not retrieved; no RECTCOL.EXE or third-party code executed.',
        'Canterbury item copyright is reserved for study/research; no redistribution grant for its fulltext is established.',
        'Full constitutive curves, command history, force/PDelta processing and damage-scope model adequacy remain open.',
    ],
    prior_inventory_sha256=digest(raw), bound_prior_files=bindings,
    structural_solver_calls=0, training_fits=0, training_admitted=False,
    independent_physical_validation=False,
    wall_seconds=time.perf_counter()-started,
)
(root / 'correspondence.json').write_text(json.dumps(summary, indent=2) + '\n')
shutil.copyfile(__file__, root / 'audit.py')
print(json.dumps({k:summary[k] for k in ['numeric_metadata_comparisons','structural_solver_calls','training_fits','wall_seconds']}))
